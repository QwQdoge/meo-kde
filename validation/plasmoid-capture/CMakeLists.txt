cmake_minimum_required(VERSION 3.21)
project(MeoPlasmoidCapture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick Qml)
find_package(KF6Package REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Kirigami REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
add_executable(meo-plasmoid-capture main.cpp)
target_link_libraries(meo-plasmoid-capture PRIVATE Qt6::Widgets Qt6::Quick Qt6::Qml Plasma::Plasma Plasma::PlasmaQuick KF6::Package)
